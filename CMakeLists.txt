cmake_minimum_required(VERSION 3.20)
project(GameboyEmulator LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Qt6 REQUIRED COMPONENTS Widgets)

add_library(video src/video.cpp)
target_include_directories(video PUBLIC src)
add_library(screen src/screen.cpp)
target_link_libraries(screen PUBLIC Qt6::Widgets video)
add_executable(gameboy src/gameboy.cpp src/Z80.cpp)
target_link_libraries(gameboy PRIVATE screen)

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(test_video tests/test_video.cpp)
    target_link_libraries(test_video PRIVATE video)
    add_test(NAME video COMMAND test_video)
    add_executable(test_screen tests/test_screen.cpp)
    target_link_libraries(test_screen PRIVATE screen)
    add_test(NAME screen COMMAND test_screen)
    set_tests_properties(screen PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    add_test(NAME cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py $<TARGET_FILE:gameboy>)
endif()
