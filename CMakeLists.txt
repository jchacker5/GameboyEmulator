cmake_minimum_required(VERSION 3.16)
project(GameboyRenderer LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BUILD_GUI "Build Qt viewer" ON)
add_library(renderer modern/renderer.cpp)
target_include_directories(renderer PUBLIC modern)
enable_testing()
add_executable(renderer_tests modern/tests.cpp)
target_link_libraries(renderer_tests PRIVATE renderer)
add_test(NAME renderer_tests COMMAND renderer_tests)
if(BUILD_GUI)
  find_package(Qt6 REQUIRED COMPONENTS Widgets)
  add_executable(gameboy_viewer modern/main.cpp)
  target_link_libraries(gameboy_viewer PRIVATE renderer Qt6::Widgets)
  add_test(NAME gui_smoke COMMAND gameboy_viewer --smoke)
  set_tests_properties(gui_smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)
endif()
